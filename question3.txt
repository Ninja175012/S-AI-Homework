#!/bin/bash

set -e

# Enter the breach_response directory
cd breach_response

# 1. Copy log_dump.txt into breach_response/
cp evidence/log_dump.txt ./log_dump.txt

# 2. Temporarily add write permission and move evidence/ into secure/
chmod u+w evidence
mv evidence secure/

# Restore evidence directory permissions
chmod 555 secure/evidence

# 3. Append "Breach confirmed" to the incident summary
chmod u+w reports/incident_summary.txt
echo "Breach confirmed" >> reports/incident_summary.txt

# Restore incident summary permissions
chmod 444 reports/incident_summary.txt

# 4. Create the forensics report and set permissions to 600
echo "Forensics Complete" > secure/forensics_report.txt
chmod 600 secure/forensics_report.txt

# 5. Append "Escalated to security team" to the incident summary
chmod u+w reports/incident_summary.txt
echo "Escalated to security team" >> reports/incident_summary.txt

# Restore incident summary permissions
chmod 444 reports/incident_summary.txt

# 6. Verify final permissions
echo "----- Final Permissions -----"
stat -c '%a %n' classified.txt secure/evidence reports/incident_summary.txt secure/forensics_report.txt

# 7. Display final incident summary
echo "----- Incident Summary -----"
cat reports/incident_summary.txt

echo "----- Task completed successfully -----"

Temporarily modify permissions to copy and move restricted files, append the required report entries, create the forensics report with specified permissions, and restore all original security permissions without modifying classified.txt
